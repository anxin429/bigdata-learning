## Day 7 · 词频统计参考实现

```python
import csv, re, sys
from collections import Counter

STOPWORDS = {"the", "a", "is", "and", "or", "of", "to", "in", "that", "with",
             "for", "on", "as", "be", "by", "are", "from", "an", "it", "may"}

def read_file(path):
    try:
        with open(path, encoding="utf-8") as f:
            return f.read()
    except FileNotFoundError:
        print(f"文件不存在：{path}"); return None

def clean_text(text):
    return re.sub(r"[^a-z\s]", "", text.lower())

def tokenize(text, stopwords):
    return [w for w in text.split() if w and w not in stopwords]

def count_words(words):
    return Counter(words)

def print_top(counter, n=20):
    for i, (w, c) in enumerate(counter.most_common(n), 1):
        print(f"{i:>3}. {w:<15} {c:>5}")

def write_csv(counter, path="wordcount_result.csv"):
    with open(path, "w", newline="", encoding="utf-8") as f:
        w = csv.writer(f); w.writerow(["word", "count"])
        w.writerows(counter.most_common())

def main():
    src = sys.argv[1] if len(sys.argv) > 1 else "sample.txt"
    text = read_file(src)
    if text is None: return
    words = tokenize(clean_text(text), STOPWORDS)
    cnt = count_words(words)
    print_top(cnt)
    write_csv(cnt)
    print(f"总词数 {len(words)}，去重后 {len(cnt)}，平均词长 {sum(map(len, words))/len(words):.2f}")
```
